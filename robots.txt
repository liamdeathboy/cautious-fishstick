# robots.txt for schplay.com — free unblocked games
# All real pages are crawlable. Below we only keep crawl budget off
# build artifacts and utility directories that should never rank.

User-agent: *
Allow: /

# Node build scripts and error pages (not content)
Disallow: /error/
Disallow: /scripts/

# Raw game builds. Every game's landing page is /games/<slug>.html; the files it
# embeds live in /games/<slug>/... (index.html, engine bundles, ROMs). Google was
# crawling those folders as hundreds of thin, untitled "pages" (/games/run3/,
# /games/vex/vex7/, /games/FNF/index.html ...) and filing them under "Crawled -
# currently not indexed", wasting crawl budget meant for the real game pages.
Disallow: /games/*/
# ...but keep game thumbnails crawlable (used as og:image and in Google Images).
Allow: /games/*.jpg$
Allow: /games/*.jpeg$
Allow: /games/*.png$
Allow: /games/*.webp$
Allow: /games/*.gif$
Allow: /games/*.avif$

Sitemap: https://schplay.com/sitemap.xml
